```bash
# 启动本地服务（Seq，以及有对应附加组件时的 PostgreSQL 与 RabbitMQ）
docker compose up -d

# 启动 Api 与 Worker（各开一个终端）
dotnet run --project src/ProductName.Api
dotnet run --project src/ProductName.Worker  # @@worker

# 全部测试
dotnet test

# 单一测试项目
dotnet test --project tests/ProductName.Domain.UnitTests

# 按名称筛选测试
dotnet test --filter-method "*CreateSample*"
```
