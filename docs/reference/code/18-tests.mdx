# 18. Tests \{#18-tests\}

## 18.1 LayerTests.cs \{#18-1-layertests-cs\}

```csharp
using System.Reflection;
using System.Text.RegularExpressions;
using MediatR;
using NetArchTest.Rules;
using ProductName.Application.Abstractions.Messaging;

namespace ProductName.Architecture.Tests;

public class LayerTests
{
    private const string Application = "ProductName.Application";
    private const string Infrastructure = "ProductName.Infrastructure";
    private const string Api = "ProductName.Api";
    private const string Worker = "ProductName.Worker"; // @@worker

    private static readonly Assembly DomainAssembly =
        typeof(ProductName.Domain.Abstractions.Result).Assembly;

    private static readonly Assembly ApplicationAssembly =
        typeof(ProductName.Application.DependencyInjection).Assembly;

    [Fact]
    public void Domain_Should_NotDependOn_OtherLayersOrEfCore()
    {
        var result = Types.InAssembly(DomainAssembly)
            .ShouldNot()
            .HaveDependencyOnAny(Application, Infrastructure, Api, Worker, "Microsoft.EntityFrameworkCore") // @@worker
            .HaveDependencyOnAny(Application, Infrastructure, Api, "Microsoft.EntityFrameworkCore") // @@!worker
            .GetResult();

        Assert.True(result.IsSuccessful);
    }

    [Fact]
    public void Application_Should_NotDependOn_InfrastructureOrPresentation()
    {
        var result = Types.InAssembly(ApplicationAssembly)
            .ShouldNot()
            .HaveDependencyOnAny(Infrastructure, Api, Worker, "MassTransit", "Serilog") // @@worker
            .HaveDependencyOnAny(Infrastructure, Api, "MassTransit", "Serilog") // @@!worker
            .GetResult();

        Assert.True(result.IsSuccessful);
    }

    [Fact]
    public void Handlers_Should_BeSealedAndInternal()
    {
        var result = Types.InAssembly(ApplicationAssembly)
            .That()
            .AreClasses()
            .And()
            .HaveNameEndingWith("Handler")
            .Should()
            .BeSealed()
            .And()
            .NotBePublic()
            .GetResult();

        Assert.True(result.IsSuccessful);
    }

    [Fact]
    public void Features_Should_NotDependOn_OtherFeatures()
    {
        var featuresNamespace = $"{Application}.Features";

        var features = ApplicationAssembly.GetTypes()
            .Select(t => t.Namespace)
            .Where(ns => ns is not null && ns.StartsWith($"{featuresNamespace}."))
            .Select(ns => ns![(featuresNamespace.Length + 1)..].Split('.')[0])
            .Distinct()
            .ToList();

        foreach (var feature in features)
        {
            var otherFeatures = features
                .Where(f => f != feature)
                .Select(f => $"{featuresNamespace}.{f}")
                .ToArray();

            if (otherFeatures.Length == 0)
                continue;

            var result = Types.InAssembly(ApplicationAssembly)
                .That()
                .ResideInNamespaceMatching($@"^{Regex.Escape($"{featuresNamespace}.{feature}")}(\.|$)")
                .ShouldNot()
                .HaveDependencyOnAny(otherFeatures)
                .GetResult();

            Assert.True(result.IsSuccessful, $"Feature '{feature}' depends on another feature.");
        }
    }

    [Fact]
    public void RequestHandlers_Should_ImplementOwnHandlerInterfaces()
    {
        Type[] ownInterfaces = [typeof(ICommandHandler<>), typeof(ICommandHandler<,>), typeof(IQueryHandler<,>)];

        var offenders = ApplicationAssembly.GetTypes()
            .Where(t => t is { IsClass: true, IsAbstract: false })
            .Where(t => ImplementsOpenGeneric(t, typeof(IRequestHandler<>)) || ImplementsOpenGeneric(t, typeof(IRequestHandler<,>)))
            .Where(t => !ownInterfaces.Any(i => ImplementsOpenGeneric(t, i)))
            .Select(t => t.FullName)
            .ToList();

        Assert.Empty(offenders);
    }

    private static bool ImplementsOpenGeneric(Type type, Type openGeneric) =>
        type.GetInterfaces().Any(i => i.IsGenericType && i.GetGenericTypeDefinition() == openGeneric);
}
```

- The first test also checks that Domain does not reference EF Core, keeping the domain free of database technology.
- The second test also checks that Application does not reference MassTransit or Serilog directly, so publishing always goes through `IEventBus` and logging through .NET's built in `ILogger`.
- The last test finds every feature under `Features` and checks that none references another (see [8.15](../spec/08-design-rules/15-feature-slices-vertical-slice.mdx)). The selector's regular expression stops at the namespace boundary, and NetArchTest matches dependency targets by whole namespace segments, so features whose names prefix each other (such as `Foo` and `FooBar`) are neither confused nor missed. Do not add a trailing dot to the targets: v1.1 did, and it silently switched the check off (verified in v1.0 with two such features, in both directions). A fresh template has no features, so the test passes.
- `RequestHandlers_Should_ImplementOwnHandlerInterfaces` makes sure every MediatR handler comes through `ICommandHandler` or `IQueryHandler`, so every response is `Result` or `Result<T>` and the validation pipeline's constraints always match (see [14.9](./14-application.mdx#14-9-validationpipelinebehavior-cs)). A handler that implements `IRequestHandler` directly would silently skip validation.

## 18.2 BaseTest.cs and ResultTests.cs (Domain.UnitTests) \{#18-2-basetest-cs-domain-unittests\}

```csharp
using ProductName.Domain.Abstractions;

namespace ProductName.Domain.UnitTests.Abstractions;

public abstract class BaseTest
{
    protected static T AssertDomainEventWasPublished<T>(IAggregateRoot aggregate)
        where T : IDomainEvent
    {
        var domainEvent = aggregate.DomainEvents.OfType<T>().SingleOrDefault();

        Assert.NotNull(domainEvent);

        return domainEvent;
    }
}
```

Domain test classes inherit it to confirm that aggregate methods raise the expected domain events.

`ResultTests.cs`:

```csharp
namespace ProductName.Domain.UnitTests.Abstractions;

public sealed class ResultTests
{
    private static readonly Error SomeError = Error.Failure("Test.Failure", "Something failed.");

    [Fact]
    public void Success_HasNoError()
    {
        var result = Result.Success();

        Assert.True(result.IsSuccess);
        Assert.Equal(Error.None, result.Error);
    }

    [Fact]
    public void Error_ConvertsToFailedResult()
    {
        Result result = SomeError;

        Assert.True(result.IsFailure);
        Assert.Equal(SomeError, result.Error);
    }

    [Fact]
    public void Value_ConvertsToSuccessfulResult()
    {
        Result<int> result = 42;

        Assert.True(result.IsSuccess);
        Assert.Equal(42, result.Value);
    }

    [Fact]
    public void Value_OfFailedResult_Throws()
    {
        Result<int> result = SomeError;

        Assert.Throws<InvalidOperationException>(() => result.Value);
    }

    [Fact]
    public void CreateFailure_BuildsTheExactResultType()
    {
        var result = CreateFailure<Result<int>>(SomeError);

        Assert.IsType<Result<int>>(result);
        Assert.Equal(SomeError, result.Error);
    }

    private static T CreateFailure<T>(Error error)
        where T : Result, IResultFactory<T> => T.CreateFailure(error);
}
```

- Covers the `Result` rules every handler relies on, including `IResultFactory` (13.11).
- It is also the template's first Domain test: Microsoft.Testing.Platform fails a test project that runs zero tests, so without it `dotnet test` fails on a fresh skeleton.

## 18.3 IntegrationTestWebAppFactory.cs (Api.IntegrationTests) \{#18-3-integrationtestwebappfactory-cs-api-integrationtests\}

```csharp
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using ProductName.Infrastructure.Database;
using Testcontainers.PostgreSql;
using Testcontainers.RabbitMq;

namespace ProductName.Api.IntegrationTests.Abstractions;

public sealed class IntegrationTestWebAppFactory : WebApplicationFactory<Program>, IAsyncLifetime
{
    private readonly PostgreSqlContainer _database = new PostgreSqlBuilder("postgres:17").Build();

    private readonly RabbitMqContainer _messageBroker = new RabbitMqBuilder("rabbitmq:4-management").Build();

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.UseEnvironment("Testing");
        builder.UseSetting("ConnectionStrings:Database", _database.GetConnectionString());
        builder.UseSetting("ConnectionStrings:MessageBroker", _messageBroker.GetConnectionString());
    }

    public async ValueTask InitializeAsync()
    {
        await Task.WhenAll(_database.StartAsync(), _messageBroker.StartAsync());

        using var scope = Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();
        await db.Database.MigrateAsync();
    }

    public override async ValueTask DisposeAsync()
    {
        await _database.DisposeAsync();
        await _messageBroker.DisposeAsync();
        await base.DisposeAsync();
    }
}
```

- PostgreSQL and RabbitMQ containers start and migrations run when tests begin; containers are released when tests end.
- xUnit v3: `IAsyncLifetime` methods return `ValueTask` and it extends `IAsyncDisposable`, so containers are released by overriding `DisposeAsync`.
- Testcontainers builders take the image name in the constructor; the old parameterless form is obsolete and fails the build under "warnings as errors".
- The `Testing` environment loads only the base file, so OpenAPI and startup migrations are off; migrations run here instead.
- For endpoints that require sign in, add a test authentication handler here that matches the project's scheme.
- Integration tests start their own containers, so CI needs no database service (the English guide starts PostgreSQL in CI `services`; this template does not need that).

## 18.4 BaseIntegrationTest.cs and HealthCheckTests.cs (Api.IntegrationTests) \{#18-4-baseintegrationtest-cs-api-integrationtests\}

```csharp
using MediatR;
using Microsoft.Extensions.DependencyInjection;
using ProductName.Infrastructure.Database;

namespace ProductName.Api.IntegrationTests.Abstractions;

public abstract class BaseIntegrationTest : IClassFixture<IntegrationTestWebAppFactory>, IDisposable
{
    private readonly IServiceScope _scope;

    protected BaseIntegrationTest(IntegrationTestWebAppFactory factory)
    {
        _scope = factory.Services.CreateScope();
        Sender = _scope.ServiceProvider.GetRequiredService<ISender>();
        DbContext = _scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();
        HttpClient = factory.CreateClient();
    }

    protected ISender Sender { get; }
    protected ApplicationDbContext DbContext { get; }
    protected HttpClient HttpClient { get; }

    public void Dispose()
    {
        HttpClient.Dispose();
        _scope.Dispose();
    }
}
```

`HealthCheckTests.cs`:

```csharp
using System.Net;

namespace ProductName.Api.IntegrationTests;
```

The smoke test proves the integration test base works: containers start, migrations run, the Api boots and `/health` answers without a token. It needs Docker and the MassTransit license (`MT_LICENSE` or `MT_LICENSE_PATH`, see [8.8](../spec/08-design-rules/08-configuration-environments-and-secrets.mdx)).

Integration test classes inherit `BaseIntegrationTest` and can send Commands and Queries directly with `Sender`, or test the full flow from the HTTP endpoint with `HttpClient`, then check the database with `DbContext`.

## 18.5 ValidationPipelineBehaviorTests.cs (Application.UnitTests) \{#18-5-validationpipelinebehaviortests-cs-application-unittests\}

```csharp
using FluentValidation;
using MediatR;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using NSubstitute;
using ProductName.Application.Abstractions.Messaging;
using ProductName.Domain.Abstractions;

namespace ProductName.Application.UnitTests.Abstractions.Behaviors;

// Runs the real MediatR pipeline in a real DI container. The fake requests below are test
// infrastructure, not business samples: they prove validation runs for Commands only.
public sealed class ValidationPipelineBehaviorTests
{
    private readonly ISender _sender;

    public ValidationPipelineBehaviorTests()
    {
        var testAssembly = typeof(ValidationPipelineBehaviorTests).Assembly;

        var services = new ServiceCollection();
        services.AddLogging();
        services.AddApplication(Substitute.For<IConfiguration>());
        services.AddMediatR(config => config.RegisterServicesFromAssembly(testAssembly));
        services.AddValidatorsFromAssembly(testAssembly, includeInternalTypes: true);

        _sender = services.BuildServiceProvider().GetRequiredService<ISender>();
    }

    [Fact]
    public async Task Command_WithInvalidInput_ReturnsValidationError()
    {
        var result = await _sender.Send(new FakeCommand(string.Empty), TestContext.Current.CancellationToken);

        Assert.True(result.IsFailure);
        Assert.IsType<ValidationError>(result.Error);
    }

    [Fact]
    public async Task CommandWithValue_WithInvalidInput_ReturnsValidationError()
    {
        var result = await _sender.Send(new FakeValueCommand(string.Empty), TestContext.Current.CancellationToken);

        Assert.True(result.IsFailure);
        Assert.IsType<ValidationError>(result.Error);
    }

    [Fact]
    public async Task Query_WithInvalidInput_SkipsValidation()
    {
        var result = await _sender.Send(new FakeQuery(string.Empty), TestContext.Current.CancellationToken);

        Assert.True(result.IsSuccess);
    }

    internal sealed record FakeCommand(string Name) : ICommand;

    internal sealed record FakeValueCommand(string Name) : ICommand<int>;

    internal sealed record FakeQuery(string Name) : IQuery<int>;

    internal sealed class FakeCommandHandler : ICommandHandler<FakeCommand>
    {
        public Task<Result> Handle(FakeCommand request, CancellationToken cancellationToken) =>
            Task.FromResult(Result.Success());
    }

    internal sealed class FakeValueCommandHandler : ICommandHandler<FakeValueCommand, int>
    {
        public Task<Result<int>> Handle(FakeValueCommand request, CancellationToken cancellationToken) =>
            Task.FromResult(Result.Success(1));
    }

    internal sealed class FakeQueryHandler : IQueryHandler<FakeQuery, int>
    {
        public Task<Result<int>> Handle(FakeQuery request, CancellationToken cancellationToken) =>
            Task.FromResult(Result.Success(1));
    }

    internal sealed class FakeCommandValidator : AbstractValidator<FakeCommand>
    {
        public FakeCommandValidator() => RuleFor(c => c.Name).NotEmpty();
    }

    internal sealed class FakeValueCommandValidator : AbstractValidator<FakeValueCommand>
    {
        public FakeValueCommandValidator() => RuleFor(c => c.Name).NotEmpty();
    }

    internal sealed class FakeQueryValidator : AbstractValidator<FakeQuery>
    {
        public FakeQueryValidator() => RuleFor(q => q.Name).NotEmpty();
    }
}
```

- Uses the real `AddApplication` registration, then adds the handlers and validators defined in this file, so it tests exactly the pipeline order and constraints the Api runs with.
- The first two tests confirm that `IResultFactory` builds the failure for both `Result` and `Result<T>` ([13.11](./13-domain.mdx#13-11-result-cs-and-iresultfactory-cs), [14.9](./14-application.mdx#14-9-validationpipelinebehavior-cs)); the third confirms that Queries skip validation even when a validator exists.
- The fake requests are test infrastructure, not business samples, so common rule 3 in [11.1](./11-file-overview.mdx#11-1-generating-with-an-ai-coding-agent) allows them.
